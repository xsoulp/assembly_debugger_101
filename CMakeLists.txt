cmake_minimum_required(VERSION 3.22)
project(ASM101 VERSION 1.0.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
# Qt Gui probes XKB optionally; ASM101 does not need its development package.
set(CMAKE_DISABLE_FIND_PACKAGE_XKB TRUE)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_library(asmcore src/DebuggerEngine.cpp src/MemoryModel.cpp src/RegisterModel.cpp src/AssemblyEditor.cpp src/MovAnimation.cpp)
target_link_libraries(asmcore PUBLIC Qt6::Widgets)
target_include_directories(asmcore PUBLIC src)
set(CMAKE_AUTORCC ON)
add_executable(asm101 src/main.cpp src/MainWindow.cpp resources/asm101.qrc)
target_link_libraries(asm101 PRIVATE asmcore)
target_compile_definitions(asm101 PRIVATE ASM101_VERSION="${PROJECT_VERSION}" EXAMPLE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/examples/example.asm")
enable_testing()
add_executable(integration tests/integration.cpp)
target_link_libraries(integration PRIVATE asmcore)
add_test(NAME debugger_integration COMMAND integration ${CMAKE_CURRENT_SOURCE_DIR}/tests/example.asm)
set_tests_properties(debugger_integration PROPERTIES TIMEOUT 30)
find_package(Qt6 REQUIRED COMPONENTS Test)
add_executable(gui_integration tests/gui_integration.cpp src/MainWindow.cpp)
target_link_libraries(gui_integration PRIVATE asmcore Qt6::Test)
target_compile_definitions(gui_integration PRIVATE EXAMPLE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/tests/example.asm")
add_test(NAME gui_integration COMMAND gui_integration)
set_tests_properties(gui_integration PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_executable(edge_integration tests/edge_integration.cpp)
target_link_libraries(edge_integration PRIVATE asmcore)
add_test(NAME edge_integration COMMAND edge_integration)
set_tests_properties(edge_integration PROPERTIES TIMEOUT 30)
add_executable(symbols_integration tests/symbols_integration.cpp src/MainWindow.cpp)
target_link_libraries(symbols_integration PRIVATE asmcore Qt6::Test)
target_compile_definitions(symbols_integration PRIVATE EXAMPLE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/tests/variables.asm")
add_test(NAME symbols_integration COMMAND symbols_integration)
set_tests_properties(symbols_integration PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_executable(stack_integration tests/stack_integration.cpp src/MainWindow.cpp)
target_link_libraries(stack_integration PRIVATE asmcore Qt6::Test)
target_compile_definitions(stack_integration PRIVATE EXAMPLE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/tests/stack.asm")
add_test(NAME stack_integration COMMAND stack_integration)
set_tests_properties(stack_integration PROPERTIES TIMEOUT 30 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
