cmake_minimum_required(VERSION 3.22)
project(ASM101 VERSION 1.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
# Qt Gui probes XKB optionally; ASM101 does not need its development package.
set(CMAKE_DISABLE_FIND_PACKAGE_XKB TRUE)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_library(asmcore src/DebuggerEngine.cpp src/MemoryModel.cpp src/RegisterModel.cpp src/AssemblyEditor.cpp src/MovAnimation.cpp)
target_link_libraries(asmcore PUBLIC Qt6::Widgets)
target_include_directories(asmcore PUBLIC src)
set(CMAKE_AUTORCC ON)
add_executable(asm101 src/main.cpp src/MainWindow.cpp resources/asm101.qrc)
target_link_libraries(asm101 PRIVATE asmcore)
target_compile_definitions(asm101 PRIVATE ASM101_VERSION="${PROJECT_VERSION}" EXAMPLE_PATH="${CMAKE_CURRENT_SOURCE_DIR}/examples/example.asm")
enable_testing()
add_executable(integration tests/integration.cpp)
target_link_libraries(integration PRIVATE asmcore)
add_test(NAME debugger_integration COMMAND integration ${CMAKE_CURRENT_SOURCE_DIR}/tests/example.asm)
set_tests_properties(debugger_integration PROPERTIES TIMEOUT 30)
find_package(Qt6 REQUIRED COMPONENTS Test)
add_executable(gui_integration tests/gui_integration.cpp src/MainWindow.cpp)
target_link_libraries(gui_integration PRIVATE asmcore Qt6::Test)

option(ASM101_BUILD_PACKAGE "Enable Debian release packaging" OFF)
if(ASM101_BUILD_PACKAGE)
    include(GNUInstallDirs)
    install(TARGETS asm101 RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(FILES examples/example.asm DESTINATION ${CMAKE_INSTALL_DATADIR}/asm101/examples)
    install(FILES packaging/asm101.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
    install(FILES resources/asm101.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
    install(FILES README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/asm101)

    find_program(ASM101_CONVERT_EXECUTABLE NAMES convert magick REQUIRED)
    set(icon_files)
    foreach(size 48 64 128 256)
        set(icon "${CMAKE_CURRENT_BINARY_DIR}/icons/${size}x${size}/asm101.png")
        add_custom_command(OUTPUT "${icon}"
            COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/icons/${size}x${size}"
            COMMAND "${ASM101_CONVERT_EXECUTABLE}" -background none "${CMAKE_CURRENT_SOURCE_DIR}/resources/asm101.svg" -resize ${size}x${size} "${icon}"
            DEPENDS resources/asm101.svg VERBATIM)
        list(APPEND icon_files "${icon}")
        install(FILES "${icon}" DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/${size}x${size}/apps)
    endforeach()
    add_custom_target(package_icons ALL DEPENDS ${icon_files})

    set(CPACK_GENERATOR DEB)
    set(CPACK_PACKAGE_NAME asm101)
    set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
    set(CPACK_PACKAGE_CONTACT "Luis Pinto <lmpinto@fc.ul.pt>")
    set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Native visual x86-64 assembly debugger for teaching")
    set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
    set(CPACK_STRIP_FILES TRUE)
    set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
    set(CPACK_DEBIAN_PACKAGE_RELEASE 1)
    set(CPACK_DEBIAN_PACKAGE_SECTION education)
    set(CPACK_DEBIAN_PACKAGE_PRIORITY optional)
    set(CPACK_DEBIAN_PACKAGE_HOMEPAGE "https://github.com/xsoulp/assembly_debugger_101")
    set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
    set(CPACK_DEBIAN_PACKAGE_DEPENDS "nasm, binutils, gdb, qt6-qpa-plugins, desktop-file-utils, gtk-update-icon-cache, hicolor-icon-theme")
    set(CPACK_DEBIAN_PACKAGE_CONTROL_EXTRA "${CMAKE_CURRENT_SOURCE_DIR}/packaging/postinst;${CMAKE_CURRENT_SOURCE_DIR}/packaging/postrm")
    set(CPACK_DEBIAN_PACKAGE_CONTROL_STRICT_PERMISSION TRUE)
    include(CPack)
endif()
